Семинар: «Тестирование 2-3 и   В деревьев»

Цель: Выяснить разницу в поведении 2-3- и B- деревьев различных конфигураций на одних и тех же операциях, оценить асимптотику и аргументировать выбор структуры данных.

Формат: 6  команд. Все команды используют один и тот же тестовый набор из  семинара 11. Каждая команда реализует свою структуру, заполняет общую таблицу и выбирает пару для сравнения деревьев.

Задание для команд:

I. Реализовать дерево, поддерживающее:

1. Вставку: Node* insert(Node* root, int key)

2. Удаление: Node* remove(Node* root, int key)

3. Поиск: Node* find(Node* root, int key)

4. Статистику: void printStats() — высота, число расщеплений/слияний/перераспределений, число узлов.

5. Для B+ — диапазонный запрос rangeQuery(lo, hi).

Примеры  базовых структур для деревьев:

// Узел B-дерева порядка t.

// В одном узле хранится от (t-1) до (2t-1) ключей

// и от t до 2t потомков (у внутреннего узла потомков на 1 больше, чем ключей).

struct Node {

    bool leaf;                  // true — узел является листом (нет потомков);

                                // false — внутренний узел (есть дети)

    int n;                      // текущее число ключей в узле (0 <= n <= 2t-1)

                                // для некорневого узла: t-1 <= n <= 2t-1

    int keys[2*T - 1];          // массив ключей, хранятся в возрастающем порядке:

                                // keys[0] < keys[1] < ... < keys[n-1]

                                // незанятые ячейки [n .. 2t-2] не используются

    Node* children[2*T];        // массив указателей на детей:

                                // у внутреннего узла используется children[0..n]

                                // (n+1 потомок), у листа — не используется (nullptr)

                                // дети упорядочены: все ключи в children[i]

                                // меньше keys[i], но больше keys[i-1]

 

    // статические счётчики операций

    static long long splits;        // число выполненных расщеплений (splitChild):

                                    // узел переполнился (2t-1 ключей) и был разделён на два

 

    static long long merges;        // число выполненных слияний (merge):

                                    // при удалении осталось t-1 ключей и узел  сливается с соседним узлом

    static long long redistributes; // число перераспределений (borrowFromPrev/Next):

                                    // при удалении ключ забирается  у соседа вместо слияния

};

// Узел B+ дерева порядка t.

// Внутренние узлы хранят только разделители (копии ключей) для навигации.

// Все  ключи (данные) хранятся  в листьях.

// Листья связаны в двусвязный список: leaf->next / leaf->prev.

struct Node {

    bool leaf;                  // true — узел является листом (хранит реальные ключи);

                                // false — внутренний узел (хранит только разделители)

 

    int n;                      // текущее число ключей/разделителей в узле

                                // (0 <= n <= 2t-1); для некорневого: t-1 <= n <= 2t-1

 

    int keys[2*T - 1];          // у листа:  ключи, возрастающие;

                                // у внутреннего узла: разделители — копии ключей,

                                //   keys[i] = минимальный ключ в поддереве children[i+1]

                                //   (используется только для навигации при поиске)

    Node* children[2*T];        // у внутреннего узла: указатели на детей

                                //   (children[0..n], n+1 потомок);

                                // у листа: не используется (nullptr)

 

    Node* next;                 //  у листа: указатель на следующий лист

                                // в связном списке (nullptr у последнего листа);

                                // ключевое поле для rangeQuery — позволяет

                                // последовательно читать листья без возврата вверх

 

    Node* prev;                 // у листа: указатель на предыдущий лист

                                // (nullptr у первого листа); опционально,

                                // нужен для обратного обхода диапазона

 

    // ---- статические счётчики операций (общие для всех узлов класса) ----

    static long long splits;        // число расщеплений узлов (splitChild)

    static long long merges;        // число слияний при удалении (merge)

    static long long redistributes; // число перераспределений ключей (borrowFromPrev/Next)

};
